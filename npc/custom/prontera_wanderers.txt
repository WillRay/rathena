//===== rAthena Script =======================================
//= Prontera Wanderers
//===== Description: =========================================
//= Ambient NPCs that patrol waypoint loops around Prontera
//= and chatter on their own schedule. Clicking one opens a
//= normal dialogue window.
//=
//= Structure:
//=   - One controller NPC (PronWander#ctrl) owns all movement
//=     and chatter state in parallel arrays and drives each
//=     wanderer by name. 'duplicate' cannot be used here --
//=     duplicates SHARE the original's NPC variables, so every
//=     copy would chase the same waypoint.
//=   - Each wanderer block below is just dialogue + appearance.
//=   - Routes and line pools are variable length. .wx/.wy and
//=     .line$ are flat arrays shared by everyone; .w_cnt and
//=     .l_cnt say how many entries each wanderer owns, and the
//=     controller derives the start offsets at load time.
//=
//=   - A .w_cnt of 0 makes an NPC static: it never moves.
//=     A .l_cnt of 0 means it never speaks on its own, which is
//=     what group members use -- they talk via conversations.
//=   - NPCs walking together do NOT stay together. Each has its
//=     own route, and two routes of different lengths drift
//=     apart even at identical speed; a speaker pausing while
//=     the others walk on makes it cumulative. A real group
//=     walk would need all members on one identical waypoint
//=     list at staggered start indices, plus pausing the whole
//=     group when any member speaks. Until that exists, keep
//=     walking groups to a single NPC.
//=   - Conversations (.g_*) are per-group scripted exchanges:
//=     the group speaks its lines in order, one every
//=     .conv_gap seconds, rests .conv_rest, then repeats.
//=     Speakers are unique NPC names, resolved to .name$
//=     indices once at load time.
//=
//= To add another wanderer:
//=   1. Add its unique name to .name$
//=   2. Add its x/y pairs to .wx / .wy, and the count to .w_cnt
//=      (or 0 in .w_cnt and no pairs, for a static NPC)
//=   3. Add its lines to .line$, and the count to .l_cnt
//=   4. Write its NPC block at the bottom
//=
//= To add a conversation:
//=   1. Add the line count to .g_cnt
//=   2. Add one speaker name per line to .g_who$
//=   3. Add the matching lines to .g_text$
//=
//= Keep every array in the same NPC order. The totals in .w_cnt,
//= .l_cnt and .g_cnt must match the flat array lengths, and every
//= .g_who$ name must exist in .name$ -- if not, the controller
//= refuses to start and says which array is wrong in the
//= map-server console.
//=
//= Notes:
//= - The sprite MUST be a player job class, not an NPC sprite
//=   (4_M_01 etc). The client has no walk animation for
//=   NPC_EVT_TYPE units, so NPC sprites slide instead of
//=   walking. With a job sprite the server respawns the unit
//=   as PC_TYPE while it moves and the animation plays.
//=   Requires PACKETVER >= 20170726 (this server: 20250319).
//= - That respawn (unit_refresh) is a clearunit + spawn, which
//=   ALSO destroys any chat bubble over the NPC's head. It
//=   fires on walk start and on walk arrival, so a wanderer
//=   must stand still while it talks -- never issue a walk in
//=   the same tick as npctalk. npcstop does not refresh.
//= - Walking is continuous because the next target is issued a
//=   few cells BEFORE the current one is reached. unit_walktoxy
//=   (unit.cpp:894) only updates the destination when a walk is
//=   already running -- it does not stop the unit -- so the NPC
//=   flows into the next leg. Waiting for it to go idle instead
//=   costs a poll-length pause at every waypoint plus an
//=   arrival refresh that pops the sprite. See .arrive.
//= - Each walk command is capped at MAX_WALKPATH (32) cells.
//=   The controller breaks longer legs into .hop-sized steps,
//=   so waypoints can be far apart -- but the interpolated
//=   midpoints are a straight line, with no knowledge of the
//=   map. If a midpoint lands in a building the walk fails and
//=   the NPC stalls there. Long legs across built-up ground
//=   need real intermediate waypoints instead.
//= - HAIRCOLOR / CLOTHCOLOR ARE A CRASH RISK. A non-zero
//=   colour makes the client load data\palette\... for that
//=   exact job+sex+index. If the .pal is missing the client
//=   does not fail gracefully -- it throws a CPaletteRes
//=   "Cannot find File" box and then dies with an access
//=   violation the moment the NPC comes into view. Index 0
//=   uses the sprite's built-in palette and never touches a
//=   file, so it is always safe. Both are 0 here on purpose;
//=   see the note on F_PronWanderLook before changing them.
//= - NPCs are named "Display::unique", NOT "Display#hidden".
//=   npc_parsename (npc.cpp:3673) only ever splits on "::" --
//=   hiding the part after a "#" is a CLIENT convention, and
//=   the client applies it to NPC-type units but not to
//=   player-type ones. Since a walking NPC is sent as PC_TYPE
//=   (see the job-sprite note above), a "#" name shows in full
//=   the whole time it is moving. "::" splits server-side, so
//=   only the display half is ever sent.
//=   .name$ and .g_who$ hold the UNIQUE half (pw_*); the
//=   display half is read back via strnpcinfo for chat prefixes.
//= - Every waypoint in this file has been checked against
//=   db/import/map_cache.dat and is walkable. If you hand-edit
//=   coordinates, re-check them -- an unwalkable waypoint fails
//=   silently and the NPC just stands there.
//============================================================

	.conv_gap  = 6;
	.conv_rest = 50;

	setarray .g_text$[0],

		"You may not. Your father hasn't eaten.",
		"...I'm keeping watch.";

	// Derive where each wanderer's waypoints and lines start in
	// the flat arrays, from the per-NPC counts above.
	//
	// Also cache the visible part of each name (before the '#')
	// so lines can be prefixed "Name: message" -- npctalk does
	// not prepend the speaker's name itself -- and stagger the
	// first line so they don't all speak at once.
	.@now = gettimetick(2);
	.@wsum = 0;
	.@lsum = 0;
	for (.@i = 0; .@i < .n; .@i++) {
		.w_off[.@i] = .@wsum;
		.@wsum += .w_cnt[.@i];
		.l_off[.@i] = .@lsum;
		.@lsum += .l_cnt[.@i];

		.disp$[.@i]     = strnpcinfo(1, getnpcid(0, .name$[.@i]));
		.next_talk[.@i] = .@now + rand(5, .talk_max);

		// -1 means "no target issued yet", so the first poll
		// starts the NPC walking.
		.tgt_x[.@i] = -1;
		.tgt_y[.@i] = -1;
	}

	// Desynced arrays would silently give someone else's route
	// or dialogue, so fail loudly instead of guessing.
	.ready = 1;
	if (.@wsum != getarraysize(.wx) || .@wsum != getarraysize(.wy)) {
		debugmes "PronWander: .w_cnt totals " + .@wsum + " but .wx has " +
			getarraysize(.wx) + " and .wy has " + getarraysize(.wy) + ". Disabled.";
		.ready = 0;
	}
	if (.@lsum != getarraysize(.line$)) {
		debugmes "PronWander: .l_cnt totals " + .@lsum + " but .line$ has " +
			getarraysize(.line$) + ". Disabled.";
		.ready = 0;
	}

	// name to its .name$ index once instead of every line.
	.@gsum = 0;
	for (.@k = 0; .@k < .g_n; .@k++) {
		.g_off[.@k]  = .@gsum;
		.@gsum      += .g_cnt[.@k];
		.g_idx[.@k]  = 0;
		.g_next[.@k] = .@now + rand(10, .conv_rest);
	}
	if (.@gsum != getarraysize(.g_who$) || .@gsum != getarraysize(.g_text$)) {
		debugmes "PronWander: .g_cnt totals " + .@gsum + " but .g_who$ has " +
			getarraysize(.g_who$) + " and .g_text$ has " +
			getarraysize(.g_text$) + ". Disabled.";
		.ready = 0;
	} else {
		for (.@k = 0; .@k < .@gsum; .@k++) {
			.@found = -1;
			for (.@i = 0; .@i < .n; .@i++) {
				if (.name$[.@i] == .g_who$[.@k]) {

					break;

		}
	}

OnTimer500:
	if (getmapusers(.map$) < 1) {
		setnpctimer 0;
		end;
	}

	.@now = gettimetick(2);

	for (.@i = 0; .@i < .n; .@i++) {
		.@nm$ = .name$[.@i];
		if (getmapxy(.@m$, .@x, .@y, BL_NPC, .@nm$) != 0)
			continue;

		// --- standing still so a bubble can be read ----
		if (.pause_until[.@i] > .@now)
			continue;

		// --- scheduled chatter -------------------------
		// Purely on a timer, not triggered by players.
		// Must not be followed by a walk this tick.
		if (.l_cnt[.@i] > 0 && .next_talk[.@i] <= .@now) {
			npcstop .@nm$;
			if (rand(100) < .own_line_chance)
				.@say$ = .line$[.l_off[.@i] + rand(.l_cnt[.@i])];
			else
				.@say$ = .common$[rand(.common_count)];
			npctalk .disp$[.@i] + ": " + .@say$, .@nm$;

			.next_talk[.@i]   = .@now + rand(.talk_min, .talk_max);
			.pause_until[.@i] = .@now + .talk_hold;
			continue;
		}

		// --- static NPCs never move --------------------
		if (.w_cnt[.@i] < 1)
			continue;

		// --- issue the next target BEFORE arriving -----
		// unit_walktoxy (unit.cpp:894) checks whether a walk is
		// already running: if it is, it only updates the
		// destination and returns, without stopping the unit.
		// So retargeting a few cells early flows straight into
		// the next leg -- no idle gap, and no arrival refresh
		// to make the sprite pop.
		//
		// The issued target is sticky: nothing is sent until the
		// NPC gets within .arrive of it, otherwise the hop
		// interpolation below would compute a slightly different
		// point every poll and re-issue constantly.
		.@need = 0;
		if (.tgt_x[.@i] < 0)
			.@need = 1;												// nothing issued yet
		else if (distance(.@x, .@y, .tgt_x[.@i], .tgt_y[.@i]) <= .arrive)
			.@need = 1;												// close enough, move on
		else if (.@x == .last_x[.@i] && .@y == .last_y[.@i])
			.@need = 1;												// stalled, or just finished talking

		.last_x[.@i] = .@x;
		.last_y[.@i] = .@y;
		if (!.@need)
			continue;

		.@b = .w_off[.@i];
		.@wpx = .wx[.@b + .idx[.@i]];
		.@wpy = .wy[.@b + .idx[.@i]];

		// Near the waypoint itself? Take the next one. This cuts
		// the corner by up to .arrive cells, which reads as a
		// person rounding a bend rather than stopping on a mark.
		if (distance(.@x, .@y, .@wpx, .@wpy) <= .arrive) {
			.idx[.@i] = (.idx[.@i] + 1) % .w_cnt[.@i];
			.@wpx = .wx[.@b + .idx[.@i]];
			.@wpy = .wy[.@b + .idx[.@i]];
		}

		// --- break long legs into hops -----------------
		// Diagonal moves cost one step, so the step count for a
		// leg is max(|dx|,|dy|). Anything over .hop is walked
		// toward an interpolated point instead.
		.@tx = .@wpx;
		.@ty = .@wpy;
		.@dx = .@tx - .@x;
		.@dy = .@ty - .@y;

		.@adx = .@dx; if (.@adx < 0) .@adx = -.@adx;
		.@ady = .@dy; if (.@ady < 0) .@ady = -.@ady;
		.@dist = .@adx; if (.@ady > .@dist) .@dist = .@ady;

		if (.@dist > .hop) {
			.@tx = .@x + (.@dx * .hop / .@dist);
			.@ty = .@y + (.@dy * .hop / .@dist);
		}

		npcwalkto .@tx, .@ty, .@nm$;
		.tgt_x[.@i] = .@tx;
		.tgt_y[.@i] = .@ty;
	}

	// --- group conversations -----------------------------
	// One line at a time, in order. The speaker stops so its
	// bubble survives the walk-refresh; the rest of the group
	// carries on. .conv_gap is deliberately longer than
	// .talk_hold so a walker gets a step or two between lines.
	for (.@k = 0; .@k < .g_n; .@k++) {
		if (.g_next[.@k] > .@now)
			continue;

		.@li  = .g_off[.@k] + .g_idx[.@k];
		.@wi  = .g_widx[.@li];
		.@wn$ = .name$[.@wi];

		npcstop .@wn$;
		npctalk .disp$[.@wi] + ": " + .g_text$[.@li], .@wn$;
		.pause_until[.@wi] = .@now + .talk_hold;

		.g_idx[.@k] = .g_idx[.@k] + 1;
		if (.g_idx[.@k] >= .g_cnt[.@k]) {
			.g_idx[.@k]  = 0;
			.g_next[.@k] = .@now + .conv_rest;
		} else {
			.g_next[.@k] = .@now + .conv_gap;
		}
	}

	setnpctimer 0;
	end;
}

//============================================================
//= Appearance helper
//= Job sprites spawn bald and undressed, so each wanderer
//= sets its own look. Runs in the calling NPC's context.
//=
//= args: <sex>, <hairstyle>, <haircolor>, <clothcolor>, <speed>
//=
//= haircolor and clothcolor are 0 for every wanderer because a
//= missing palette file CRASHES the client (see header). To add
//= colour, verify the .pal actually ships in your GRF first --
//= open data.grf in GRF Editor and look under:
//= colour, verify the .pal actually ships in your GRF first.
//= Open data.grf in GRF Editor under \palette\. The folder and
//= filename parts are Korean; romanised they are:
//=   palette\[mom]\<job>_[nam|yeo]_<n>.pal            body
//=   palette\[meori]\[meori]<style>_[nam|yeo]_<n>.pal hair
//= where [mom]=body, [meori]=head, [nam]=male, [yeo]=female.
//= Colour indices are NOT uniform across jobs -- one that exists
//= for one job may be absent for another, so check per job+sex,
//= and test each change by walking up to that specific NPC.
//= uniform across jobs -- a colour that exists for one job may
//= be absent for another, so check per job+sex, and test each
//= change by walking up to that specific NPC.
//============================================================
function	script	F_PronWanderLook	{
	.@gid = getnpcid(0);
	setunitdata .@gid, UNPC_SEX,        getarg(0);
	setunitdata .@gid, UNPC_HAIRSTYLE,  getarg(1);
	setunitdata .@gid, UNPC_HAIRCOLOR,  getarg(2);
	setunitdata .@gid, UNPC_CLOTHCOLOR, getarg(3);
	setunitdata .@gid, UNPC_HEADTOP,    0;
	npcspeed getarg(4, 190);
	return;
}
