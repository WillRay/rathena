OnMenu:
	// Which shelves did the NPC that called us expose? No argument = all.
	deletearray @qs_tier[0], getarraysize(@qs_tier);
	set .@n, getarraysize(@i);
	if (!.@n) {
		for (set .@k,1; .@k<=.NumShops; set .@k,.@k+1)
			set @qs_tier[.@k-1], .@k;
	} else {
		for (set .@k,0; .@k<.@n; set .@k,.@k+1)
			set @qs_tier[.@k], @i[.@k];
	}
	deletearray @i[0], getarraysize(@i);
	set .@n, getarraysize(@qs_tier);
	if (!.@n) {
		mes "[Rare Hat Merchant]";
		mes "I have nothing to trade just now.";
		close;
	}

	mes "[Rare Hat Merchant]";
	mes "Rare headgear, straight trade -- bring me the materials and the hat is yours.";
	mes " ";
	mes "Which shelf would you like to see?";
	next;
	set .@m$,"";
	for (set .@k,0; .@k<.@n; set .@k,.@k+1)
		set .@m$, .@m$ + .Shops$[@qs_tier[.@k]] + ":";
	set .@m$, .@m$ + "^777777Nothing, thanks^000000";
	set .@s, select(.@m$);
	if (.@s > .@n) close;
	set .@t, @qs_tier[.@s-1];

	close2;
	if (!callshop("qshop_t"+.@t)) {
		mes "[Rare Hat Merchant]";
		mes "^FF0000My shop window will not open.^000000";
		mes " ";
		mes "The barter data for that shelf is missing. An admin needs to run ^0055FFperl tools/quest_shop_barter.pl^000000 and reload.";
		close;
	}
	end;

function	Add	{
	if (getitemname(getarg(1)) == "null") {
		debugmes "[quest_shop] reward #"+getarg(1)+" is not in the item database (skipped).";
		return;
	}
	setarray .@j[0],getarg(2),getarg(3),getarg(4);
	for(set .@i,5; .@i<getargcount(); set .@i,.@i+2) {
		if (getitemname(getarg(.@i)) == "null") {
			debugmes "[quest_shop] requirement #"+getarg(.@i)+" of reward #"+getarg(1)+" is not in the item database (whole recipe skipped).";
			return;
		}
		setarray .@j[.@i-2],getarg(.@i),getarg(.@i+1);
	}
	copyarray getd(".q_"+getarg(1)+"[0]"),.@j[0],getarraysize(.@j);
	// Per-shelf index, so the boot check can walk a shelf.
	set .@n, getarraysize(getd(".list"+getarg(0)));
	setd ".list"+getarg(0)+"["+.@n+"]", getarg(1);
	return;
}
}

function	script	qshop	{
	deletearray @i[0],getarraysize(@i);
	for(set .@i,0; .@i<getargcount(); set .@i,.@i+1)
		set @i[.@i],getarg(.@i);
	doevent "quest_shop::OnMenu";
	end;
}
