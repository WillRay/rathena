//===== rAthena Script =======================================
//= mag_dun03 - Nogg Road 3F entrance
//===== Description: =========================================
//= Adapted from npc/re/warps/dungeons/mag_dun.txt.
//=
//= Two changes from the renewal original:
//=
//=  1. The entrance moves to mag_dun02 246,70. The area is
//=     empty in pre-renewal - npc/warps/dungeons/mag_dun.txt
//=     has nothing near it and the nearest objects on the map
//=     are at 46,40 and 43,45 on the far west side.
//=
//=  2. The gate is TRANSCENDENT CLASS, not level 175. The
//=     eaclass()&EAJL_UPPER test covers High Novice through
//=     the trans 2nd classes and excludes everyone else.
//=     There is deliberately no level requirement on the door
//=     itself - the monsters enforce that on their own.
//============================================================

mag_dun02,248,72,3	script	Republic Guard#mag03	4_M_EIN_SOLDIER,{
	mes "[Republic Guard]";
	mes "Past this point the rock never cooled. What comes out of there is not what went in.";
	next;
	mes "[Republic Guard]";
	if (eaclass()&EAJL_UPPER) {
		mes "You have already died once and come back, haven't you. I can see it.";
		mes "Go on, then. You are the only sort that walks back out.";
	} else {
		mes "You are not ready. I have watched ordinary adventurers go down there, and I have watched nobody come up.";
		mes "Come back when you have been reborn.";
	}
	close;
}

mag_dun02,246,70,0	script	mag02a_mag03a	WARPNPC,1,1,{
	end;

OnTouch:
	if (!(eaclass()&EAJL_UPPER)) {
		npctalk "Republic Guard: Stop. Only the reborn go down there.", "Republic Guard#mag03", bc_self;
		end;
	}
	warp "mag_dun03",120,120;
	end;
}

mag_dun03,124,120,0	warp2	mag03a_mag02a	1,1,mag_dun02,243,68
